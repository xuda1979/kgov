    """Read up to ``n`` messages from a topic (oldest first).

    A fresh consumer with ``auto.offset.reset=earliest`` runs under a unique
    group id per call so concurrent scans do not collide on commits. Messages
    are never committed. If ``value_decoder`` is supplied it is consulted for
    values that are not valid JSON (e.g. an :class:`AvroRecordDecoder`).
    """
    _require_kafka()
    from confluent_kafka import Consumer, KafkaError, KafkaException

    unique_group = f"{group_id}-{id(object()):x}"
    conf = {
        "bootstrap.servers": bootstrap,
        "group.id": unique_group,
        "auto.offset.reset": "earliest",
        "enable.auto.commit": False,
    }
    consumer = Consumer(conf)
    out: list[BrokerMessage] = []
    try:
        consumer.subscribe([topic])
        deadline = _now_ms() + timeout * 1000
        while _now_ms() < deadline:
            msg = consumer.poll(timeout=5.0)
            if msg is None:
                continue
            if msg.error():
                if msg.error().code() == KafkaError._PARTITION_EOF:
                    continue
                if msg.error().code() == KafkaError._TRANSPORT:
                    raise KafkaException(msg.error())
                continue
            ts = msg.timestamp()
            out.append(
                BrokerMessage(
                    topic=msg.topic(),
                    partition=msg.partition(),
                    offset=msg.offset(),
                    key=_json_safe(msg.key()),
                    value=_decode_value(msg.value(), value_decoder),
                    timestamp_ms=ts[1] if ts else None,
                    headers=dict(msg.headers() or []),
                )
            )
            if n is not None and len(out) >= n:
                break
    finally:
        consumer.close()
    return out

    # -- lifecycle ---------------------------------------------------------

    def start(self, port: int = 0) -> str:
        """Bind on loopback (0 => OS-assigned ephemeral port); return base URL."""
        if self._server:
            return self.base_url
        handler = type("BoundHandler", (RegistryHandler,), {"registry": self})
        try:
            self._server = ThreadingHTTPServer(("127.0.0.1", port), handler)
        except OSError as exc:
            raise MockRegistryError(f"mock registry failed to bind: {exc}") from exc
        self._bound_port = int(self._server.server_address[1])
        self._thread = threading.Thread(
            target=self._server.serve_forever, daemon=True, name="mock-registry"
        )
        self._thread.start()
        return self.base_url

    def stop(self) -> None:
        if self._server is not None:
            self._server.shutdown()
            self._server.server_close()
            self._server = None
        if self._thread is not None and self._thread.is_alive():
            self._thread.join(timeout=5)
            self._thread = None
        self._bound_port = 0

    def __enter__(self) -> "MockRegistry":
        self.start()
        return self

    def __exit__(self, *exc: Any) -> None:
        self.stop()

def _parse_schema_string(schema: Any) -> Any:
    """The Confluent wire returns schema as a JSON string; normalize to a dict."""
    if schema is None:
        return None
    if isinstance(schema, (dict, list)):
        return schema
    try:
        return json.loads(str(schema))
    except json.JSONDecodeError:
        # Not parseable as JSON — keep the raw string (e.g. a bare primitive).
        return str(schema)

def _type_ok(t: str, val: Any) -> bool:
    if val is None:
        return True  # handled by non_null if that flag is set
    if t == "string":
        return isinstance(val, str)
    if t == "integer" or t == "int":
        return isinstance(val, int) and not isinstance(val, bool)
    if t == "number":
        return isinstance(val, (int, float)) and not isinstance(val, bool)
    if t == "boolean" or t == "bool":
        return isinstance(val, bool)
    if t == "array":
        return isinstance(val, list)
    if t in ("object", "record", "map"):
        return isinstance(val, dict)
    return False

def _avro_wire_bytes(schema_id: int, record: dict, avro_schema: dict) -> bytes:
    """Encode a record into the Confluent wire format (magic + id + datum)."""
    import io
    import json

    from avro import io as avro_io
    from avro import schema as avro_schema_mod

    parsed = avro_schema_mod.parse(json.dumps(avro_schema))
    buf = io.BytesIO()
    writer = avro_io.DatumWriter(parsed)
    writer.write(record, avro_io.BinaryEncoder(buf))
    return b"\x00" + schema_id.to_bytes(4, byteorder="big") + buf.getvalue()
